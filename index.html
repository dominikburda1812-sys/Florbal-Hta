import os import time # --- POMOCNÉ FUNKCE PRO LEPŠÍ DESIGN --- def vycisti_obrazovku(): """Vyčistí konzoli pro lepší vizuální zážitek ze hry.""" os.system('cls' if os.name == 'nt' else 'clear') def tiskni_nadpis(text): """Vytiskne hezky zformátovaný nadpis.""" print("=" * 40) print(f" {text.upper()} ".center(40, "=")) print("=" * 40) # --- TŘÍDY PRO HRU --- class Florbalka: def __init__(self, nazev, level): self.nazev = nazev self.level = level # Nyní má každá florbalka svůj level! class Hrac: def __init__(self, jmeno): self.jmeno = jmeno self.stav = "Připraven" # Možnosti: Připraven, Spí, Hraje, Odpočívá class Klub: def __init__(self, jmeno_klubu): self.jmeno = jmeno_klubu self.penize = 10000 # Hráč začíná s 10 000 Kč # Batoh obsahuje pouze platné objekty typu Florbalka s levelem self.batoh = [Florbalka("Základní hůl", level=1)] self.sestava = [Hrac("Jan Novák"), Hrac("Petr Svoboda")] def zobraz_info(self): tiskni_nadpis(f"Klub: {self.jmeno}") print(f"Peníze: {self.penize} Kč") print("\n--- BATOH (Florbalky) ---") if not self.batoh: print("Batoh je prázdný.") else: for f in self.batoh: print(f"- {f.nazev} (Level {f.level})") print("\n--- SESTAVA ---") for h in self.sestava: print(f"- {h.jmeno} | Stav: {h.stav}") print("-" * 40) def hraj_zapas(self): tiskni_nadpis("ZÁPAS ZAČÍNÁ") # OPRAVA: Probudíme všechny hráče v sestavě! for hrac in self.sestava: if hrac.stav == "Spí": print(f"Probouzím hráče: {hrac.jmeno}!") hrac.stav = "Hraje" print("Hráči jsou na hřišti, nikdo nespí a hrají naplno!\n") time.sleep(2) print("... probíhá zápas ...\n") time.sleep(2) print("Zápas skončil! Dobrá práce.") # Po zápase jdou hráči odpočívat for hrac in self.sestava: hrac.stav = "Odpočívá" # --- HLAVNÍ SMYČKA HRY --- def hlavni_menu(): vycisti_obrazovku() tiskni_nadpis("FLORBALOVÝ MANAŽER") # Založení klubu nazev = input("Vítej trenére! Jak se bude jmenovat tvůj klub? \n> ") muj_klub = Klub(nazev) # Herní smyčka while True: vycisti_obrazovku() muj_klub.zobraz_info() print("\nCo chceš dělat?") print("1. Hrát zápas") print("2. Poslat hráče spát (pro otestování probuzení)") print("3. Konec hry") volba = input("\nVyber číslo: ") if volba == "1": vycisti_obrazovku() muj_klub.hraj_zapas() input("\nStiskni Enter pro návrat do menu...") elif volba == "2": for h in muj_klub.sestava: h.stav = "Spí" print("Hráči teď spí. Zkus dát 'Hrát zápas' a uvidíš, že se probudí.") input("\nStiskni Enter pro návrat...") elif volba == "3": print("Díky za hru!") break else: print("Neplatná volba.") time.sleep(1) # Spuštění hry if __name__ == "__main__": hlavni_menu()
